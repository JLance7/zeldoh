cmake_minimum_required(VERSION 3.20)
project(Zeldoh LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This edition uses the Windows desktop API. Build on Windows.")
endif()
add_executable(Zeldoh WIN32 src/main.cpp)
target_compile_features(Zeldoh PRIVATE cxx_std_17)
target_compile_definitions(Zeldoh PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(Zeldoh PRIVATE gdi32 user32 gdiplus)
if(MINGW)
  target_link_options(Zeldoh PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
